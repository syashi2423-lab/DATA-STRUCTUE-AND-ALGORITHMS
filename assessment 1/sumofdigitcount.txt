import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        int sum = 0, count = 0;

        while (n > 0) {
            sum += n % 10;
            count++;
            n /= 10;
        }

        System.out.println("Sum = " + sum);
        System.out.println("Count = " + count);
    }
}
