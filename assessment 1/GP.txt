import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        double a = sc.nextDouble();
        double r = sc.nextDouble();

        for (int i = 1; i <= n; i++) {
            System.out.print(a + " ");
            a = a * r;
        }
    }
}
