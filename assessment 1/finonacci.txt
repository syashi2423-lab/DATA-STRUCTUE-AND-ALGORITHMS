import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        int a = 0, b = 1, sum = 0;

        for (int i = 1; i <= n; i++) {
            sum += a;
            a = a + b;
            b = a - b;
        }

        System.out.println(sum);
    }
}
